void matriz1(int matriz[N][N],int f,int c){
    int i,j;
    matriz[f][c] = 0;
    for(i = f+1,j=c+1;i<N && j<N;i++,j++)//diagonal inferior der
        matriz[i][j] = 0;
    for(i = f-1,j=c+1;i >= 0 && j<N;i--,j++)// diagonal superior der
        matriz[i][j] = 0;
    for(i = f-1,j=c-1;i >= 0 && j >= 0;i--,j--)// diagonal superior izq
        matriz[i][j] = 0;
    for(i = f+1,j=c-1;i<N && j>=0;i++,j--)// diagonal inferior izq
        matriz[i][j] = 0;

}

void diagonales(int matriz[N][N],int f,int c){
    int i,j;
    matriz[f][c] = 0;
    for(i = f+1,j=c+1;i<N && j<N;i++,j++)//diagonal inferior derecha
        matriz[i][j] = 0;
    for(i = f-1,j=c+1;i >= 0 && j<N;i--,j++)// diagonal derecha superior
        matriz[i][j] = 0;
    for(i = f-1,j=c-1;i >= 0 && j >= 0;i--,j--)// diagonal superior derecha
        matriz[i][j] = 0;
    for(i = f+1,j=c-1;i<N && j>=0;i++,j--)// diagonal inferior izquierda
        matriz[i][j] = 0;

}

void quitar_cad(char cadena[N],char cadena1[N]){
    int i,j;
    for(i=0; i<N; i++)
        for(j=0;j<N && cadena[i] != ' ';j++)
            if(cadena[i] == cadena1[j])
                cadena[i] = ' ';
}

void mostrar_cad(char cadena[N]){
    int i;
    for(i=0;i<N;i++)
        printf ("%c",cadena[i]);
}

void mostrar_mat(int matriz[N][N]){
    int i,j;
    for(i=0; i<N; i++) {
        for (j = 0; j < N; j++)
            printf("%i ", matriz[i][j]);
        printf("\n");
    }

}
